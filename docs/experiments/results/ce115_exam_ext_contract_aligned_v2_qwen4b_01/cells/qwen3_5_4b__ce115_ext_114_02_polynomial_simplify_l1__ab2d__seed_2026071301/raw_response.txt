    class PolynomialOps:
        @staticmethod
        def coeffs_from_py_expression(expression: str, var='x') -> List[Union[int, Fraction]]:
            """Extract coefficients from a polynomial string."""
            import sympy as sp
            x = sp.Symbol('x')
            expr = sp.sympify(expression)
            poly = sp.Poly(expr, x)
            coeffs = [c for c in poly.all_coeffs()]
            # Convert to Fraction if not int to ensure exact arithmetic representation
            return [Fraction(c) if isinstance(c, (int, float)) else c for c in coeffs]

        @staticmethod
        def to_degree_map(coeffs: List[Union[int, Fraction]]) -> Dict[str, Union[int, str]]:
            """Convert coefficients list to degree-string map."""
            # Ensure we have the correct length and order (highest degree first)
            if not coeffs or len(coeffs) == 0:
                return {}

            result = {}
            for i, coeff in enumerate(coeffs):
                deg_str = str(len(coeffs) - i)
                val = int(coeff) if isinstance(coeff, Fraction) else float(coeff) # Convert back to int/float logic but keep exactness via Fraction during calc
                
                # If it's a whole number integer, store as int. Otherwise p/q string? 
                # The spec says "exact ints or irreducible p/q strings".
                # Since coeffs_from_py_expression returns Fractions for non-ints in sympy context usually, we check:
                
                if isinstance(coeff, Fraction):
                    num = coeff.numerator
                    den = coeff.denominator
                    if den == 1:
                        result[deg_str] = int(num)
                    else:
                        # Format as p/q string. Ensure irreducible (Fraction handles this).
                        sign = '-' if num < 0 and den > 0 else ''
                        abs_num = abs(num)
                        abs_den = abs(den)
                        result[deg_str] = f"{sign}{abs_num}/{abs_den}"
                elif isinstance(coeff, int):
                    result[deg_str] = coeff
            return result
