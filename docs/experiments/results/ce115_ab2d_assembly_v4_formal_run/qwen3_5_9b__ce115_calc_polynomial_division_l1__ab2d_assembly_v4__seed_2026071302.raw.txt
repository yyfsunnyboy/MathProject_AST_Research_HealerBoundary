def generate(level=1, **kwargs):
    from core.prompts.domain_function_library import PolynomialOps, FractionOps
    
    dividend_coeffs = [-6, -4, 3]
    divisor_coeffs = [1, -2]
    
    quotient_coefficients, remainder_coefficients = PolynomialOps.div_qr(dividend_coeffs, divisor_coeffs)
    
    question_text = f"Divide the polynomial {PolynomialOps.format_plain(quotient_coefficients)} by the linear divisor defined by coefficients {divisor_coeffs}. Report the resulting quotient and remainder."
    
    return {
        "question_text": question_text,
        "correct_answer": {
            "quotient_coefficients": quotient_coefficients,
            "remainder_coefficients": remainder_coefficients
        },
        "oracle_payload": {"dividend_coefficients": dividend_coeffs, "divisor_coefficients": divisor_coeffs}
    }
