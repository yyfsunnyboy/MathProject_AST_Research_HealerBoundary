        """Creates a Fraction from string input."""
        # Handle simple decimals like "2.79" -> 279/100
        if '.' in value_str:
            parts = value_str.split('.')
            numerator = int(parts[0]) * (10 ** len(parts[1])) + int(parts[1])
            denominator = 10 ** len(parts[1])

                raise ValueError(f"Invalid rational input: {value_str}")

    def mul(a, b):
        """Multiplies two Fractions."""
        if isinstance(a, str): a = DomainAPI.create(a)
        if isinstance(b, str): b = DomainAPI.create(b)

        if isinstance(b, str): b = DomainAPI.create(b)
