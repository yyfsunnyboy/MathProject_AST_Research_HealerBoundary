# -*- coding: utf-8 -*-

def generate(level=1, **kwargs):
    candidates = [11, 12, 13, 14]
    n = 156

    def is_prime(num):
        if num <= 1:
            return False
        for i in range(2, int(num**0.5) + 1):
            if num % i == 0:
                return False
        return True

    correct_answer = None
    for candidate in candidates:
        if n % candidate == 0 and is_prime(candidate):
            correct_answer = candidate
            break

    question_text = f"Which of the following prime numbers divides {n} exactly? $\\boxed{{\\text{{candidates: {candidates}}}}}$"
    oracle_payload = {"candidates": candidates, "n": n}

    return {
        "question_text": question_text,
        "correct_answer": correct_answer,
        "oracle_payload": oracle_payload,
    }
