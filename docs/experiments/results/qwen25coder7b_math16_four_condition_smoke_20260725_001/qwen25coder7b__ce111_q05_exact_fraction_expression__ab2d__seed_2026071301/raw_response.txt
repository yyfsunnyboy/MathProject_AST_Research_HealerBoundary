    c = FractionOps.create(23, 22)

    correct_answer = {
        "numerator": final_result.numerator,
        "denominator": final_result.denominator,

    }
    
    question_text = f"$$\\frac{{9}}{{22}} + \\frac{{11}}{{18}} - (\\frac{{23}}{{22}} - \\frac{{7}}{{18}}) = {correct_answer['canonical_latex']}$$"

        "question_text": question_text,

    }
