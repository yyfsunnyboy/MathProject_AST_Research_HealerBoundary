import math

def get_prime_factors(n):
    factors = []
    d = 2
    temp_n = n
    while d * d <= temp_n:
        if temp_n % d == 0:
            count = 0
            while temp_n % d == 0:
                count += 1
                temp_n //= d
            factors.extend([(d, count)])
        d += 1
    if temp_n > 1:
        factors.append((temp_n, 1))
    return factors
