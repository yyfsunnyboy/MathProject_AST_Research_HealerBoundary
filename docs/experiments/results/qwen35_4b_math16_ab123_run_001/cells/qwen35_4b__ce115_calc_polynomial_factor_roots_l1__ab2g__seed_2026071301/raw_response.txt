def ce115_calc_polynomial_factor_roots_l1(polynomials, difficulty=1):
    import sympy
    
    def solve():
        if not polynomials or len(polynomials) != 2:
            return None
        
        coeffs = list(map(int, polys[0]))
        
        # Check for quadratic coefficient zero to avoid division by zero in later steps.
        a = -coeffs[-1] * sympy.Rational(0, 1) + 0j if len(coeffs) > 2 else 0
        
        b_n_minus_3_i_k_j_l_q_t_u_v_r_o_p_s_m_g_a_f_h_e_y_z_w_x_c_d = polys[0][-4] - int(sympy.sqrt(abs(
            sympy.nsimplify((sympy.Rational(polys[1], 2) / (a + b_n_minus_3_i_k_j_l_q_t_u_v_r_o_p_s_m_g_a_f_h_e_y_z_w_x_c_d)) ** 2 -
                           polys[0][-5] * sympy.sqrt(sympy.nsimplify(((sympy.Rational(polys[-1], -a + b_n_minus_3_i_k_j_l_q_t_u_v_r_o_p_s_m_g_a_f_h_e_y_z_w_x_c_d)) ** 2 /
                            (b_n_minus_3_i_k_j_l_q_t_u_v_r_o_p_s_m_g_a_f_h_e_y_z_w_x_c_d - polys[0][-4]))**2 + ((sympy.Rational(polys[-1], a * polys[-5] + b_n_minus_3_i_k_j_l_q_t_u_v_r_o_p_s_m_g_a_f_h_e_y_z_w_x_c_d) ** 2 /
                            (a * polys[0][-4]) - a))**2)).as_real_imag()[1])) if len(coeffs) >= 6 else sympy.Rational(0, 1).real
            
        return {'question_text': r'\text{Given } p(x)=x^3-5x+8\ \text{ and }\ q(x)=x^4-x^2+\frac{x}{z}, \text{ find the roots of each polynomial.}', 'correct_answer': None, 'oracle_payload': {}}
    
    return solve()
