class FractionOps:
    @staticmethod
    def create(value):
        if isinstance(value, str) and "/" in value:
            parts = value.split("/")
            num = int(parts[0])
            den = int(parts[1])
            return (num, den) # Returning tuple as a proxy for Fraction object logic
        elif isinstance(value, float):
            g = math.gcd(int(abs(value)), 1) if abs(value) < 2 else value.denominator
            num, den = simplify_fraction(int(round(value * 10**9)) // (10**9), 1) # Fallback logic for simplicity in mock
            return create_from_float(value)
        elif isinstance(value, tuple):
            n, d = value
            g = math.gcd(abs(n), abs(d)) if d != 0 else 1
            return simplify_fraction(n // g, d // g)
        raise ValueError("Unsupported input type for FractionOps.create")

    @staticmethod
    def add(a: tuple, b: tuple):
        n_a, d_a = a
        n_b, d_b = b
        
        # Compute LCM of denominators
        lcm_d = (d_a * d_b) // math.gcd(d_a, d_b)
        
        new_n = (n_a * (lcm_d // d_a)) + (n_b * (lcm_d // d_b))
        g = math.gcd(abs(new_n), abs(lcm_d))
        
        return simplify_fraction(new_n // g, lcm_d // g)

    @staticmethod
    def to_latex(val: tuple, mixed=False):
        n, d = val
        if d == 1:
            return f"{n}" # Integer case usually represented as integer in simple contexts unless specified otherwise
        sign = "-" if n < 0 else ""
        abs_n, abs_d = abs(n), abs(d)
        
        latex_str = r"\frac" + "{%s}{" % (abs_n if not mixed else "1") # Placeholder for strict LaTeX
        
        # Constructing proper fraction or mixed number string representation in LaTeX
        numerator_part = str(abs_n)
        denominator_part = str(abs_d)
        
        return f"{sign}\\frac{{{numerator_part}}}{{{{{denominator_part}}}}}}"

def simplify_fraction(n, d):
    if d == 0: raise ZeroDivisionError("Denominator cannot be zero")
    g = math.gcd(abs(n), abs(d))
    n //= g
    d //= g
    # Canonical form usually requires positive denominator
    if d < 0:
        n, d = -n, -d
    return (n, d)

def create_from_float(val):
    from fractions import Fraction as F
    f = F(val).limit_denominator()
    return simplify_fraction(f.numerator, f.denominator)
