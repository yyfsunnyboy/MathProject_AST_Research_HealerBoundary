class FractionOps:
    @staticmethod
    def create(value_str):
        # Parses simple fraction strings like "3/7", "-1/4", or integers
        if '/' in value_str:
            num_str, den_str = value_str.split('/')
            return int(num_str), int(den_str)
        else:
            val = float(value_str)
            sign = 1 if val >= 0 else -1
            abs_val = abs(val)
            # Find GCD for irreducible form immediately upon creation to ensure canonical state logic holds internally
            g = math.gcd(int(abs_val), 1) 
            return int(sign * (abs_val / g)), 1

    @staticmethod
    def sub(a, b):
        if isinstance(a, tuple) and len(a) == 2: a_n, a_d = a else: raise ValueError("First arg must be fraction")
        if isinstance(b, tuple) and len(b) == 2: b_n, b_d = b else: raise ValueError("Second arg must be fraction")
        
        # Subtraction rule: n1/d1 - n2/d2 = (n1*d2 - n2*d1) / (d1*d2)
        new_num = a_n * b_d - b_n * a_d
        new_den = a_d * b_d
        
        g = math.gcd(new_num, new_den)
        
        # Ensure canonical representation: positive denominator
        if new_den < 0:
            new_num = -new_num
            new_den = -new_den
            
        return (new_num // g, new_den // g), False

    @staticmethod
    def to_latex(val_tuple, mixed=False):
        num, den = val_tuple
        # Helper for GCD reduction inside latex just in case logic drifts
        common = math.gcd(num, den) if not (den == 1 and abs(num) <= 20) else 1 
        final_num = num // common
        final_den = den // common
        
        if mixed:
            whole = int(final_num / final_den)
            rem_num = abs(final_num - whole * final_den) # This logic is simplified for robustness, usually remainder
            # Simplified strict fraction latex output as per standard mathjax expectations in these tasks
            return f"\\frac{{{final_num}}}{{({{final_den}})}}" if not (den==1 and num==-20 else False)

        sign = "-" if final_num < 0 else ""
        abs_n, d_abs = abs(final_num), final_den
        
        # Optimization for small integers to render as numbers in LaTeX when denominator is 1 or very simple? 
        # The prompt asks for canonical_latex. Usually \frac{n}{d}.
        return f"{sign}\\frac{{{abs_n}}}{{({{d_abs}})}}" if d_abs != 1 else str(abs_n)
