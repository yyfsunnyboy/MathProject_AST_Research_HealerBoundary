# -*- coding: utf-8 -*-

def generate(level=1, **kwargs):
    # Define all frozen parameters directly.
    factor_order_policy = "strict_source_template"
    quadratic_coefficients = [39, 5, -14]
    template_left_x_coefficient = 3

    # Compute the exact result using native Python.
    A, B, C = quadratic_coefficients
    L = template_left_x_coefficient

    # L * b = A => b = A // L
    b = A // L

    a_sol, c_sol = None, None
    for a in range(-abs(C) - 1, abs(C) + 2):
        if a == 0:
            continue
        if C % a == 0:
            c = C // a
            if L * c + a * b == B:
                a_sol = a
                c_sol = c
                break

    if a_sol is None:
        a_sol = 2
        c_sol = -7

    correct_answer = a_sol + 2 * c_sol

    # Format the polynomial to LaTeX
    parts = []
    if A == 1:
        parts.append("x^2")
    elif A == -1:
        parts.append("-x^2")
    elif A != 0:
        parts.append(f"{A}x^2")
    
    if B > 0:
        parts.append(f"+ {B}x" if B != 1 else "+ x")
    elif B < 0:
        parts.append(f"- {-B}x" if B != -1 else "- x")
        
    if C > 0:
        parts.append(f"+ {C}")
    elif C < 0:
        parts.append(f"- {-C}")
    
    poly_latex = " ".join(parts).replace("+ -", "- ").replace("  ", " ")

    question_text = (
        f"The quadratic polynomial ${poly_latex}$ can be factored in the form "
        f"$({L}x + a)(bx + c)$ for some integers $a$, $b$, and $c$. "
        f"Find the value of $a + 2c$."
    )

    oracle_payload = {
        "factor_order_policy": factor_order_policy,
        "quadratic_coefficients": quadratic_coefficients,
        "template_left_x_coefficient": template_left_x_coefficient,
    }

    return {
        "question_text": question_text,
        "correct_answer": correct_answer,
        "oracle_payload": oracle_payload,
    }
