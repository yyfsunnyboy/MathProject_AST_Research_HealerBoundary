# -*- coding: utf-8 -*-
from core.prompts.domain_function_library import FractionOps

def generate(level=1, **kwargs):
    # Frozen parameters
    oracle_payload = {"expression": "3/7 - (-1/4)"}
    
    # Compute using FractionOps
    f1 = FractionOps.create("3/7")
    f2 = FractionOps.create("-1/4")
    res = FractionOps.sub(f1, f2)
    
    # Extract numerator and denominator
    num = res.numerator
    den = res.denominator
    
    # Format canonical LaTeX
    if num < 0:
        canonical_latex = f"-\\frac{{{-num}}}{{{den}}}"
    else:
        canonical_latex = f"\\frac{{{num}}}{{{den}}}"
        
    question_text = r"Evaluate the following expression:" + "\n" + r"\[ \frac{3}{7} - \left(-\frac{1}{4}\right) \]"
    
    correct_answer = {
        "numerator": num,
        "denominator": den,
        "canonical_latex": canonical_latex
    }
    
    return {
        "question_text": question_text,
        "correct_answer": correct_answer,
        "oracle_payload": oracle_payload
    }
