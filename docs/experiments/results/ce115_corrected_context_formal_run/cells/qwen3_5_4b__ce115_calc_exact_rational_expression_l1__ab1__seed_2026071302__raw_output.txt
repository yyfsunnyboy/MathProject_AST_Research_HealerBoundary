        if '.' in s:
            integer_part, fractional_part = s.split('.')

            reduced_num *= -1
            reduced_den *= -1
        if reduced_den == 1:
            return str(reduced_num)
        else:
            return f"{reduced_num}/{reduced_den}"

        while b != 0:
            a, b = b, a % b
