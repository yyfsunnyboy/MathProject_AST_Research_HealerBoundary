    def is_square_free(n):
        if n <= 1:
            return True

                return False

        return True

    def get_prime_factorization_sqrt_v3(n):
        """Returns coefficient and radicand for sqrt(n)."""
        temp = n
        
        # Prime factorize to count exponents of each prime p: p^e -> contributes 2*floor(e/2) to power, remainder e%2 stays.
        
        factors = {}
        d = 2
        while d * d <= temp:
            if temp % d == 0:
                cnt = 0
                while temp % d == 0:
                    cnt += 1
                    temp //= d
                factors[d] = cnt
            
            # Check next prime? Just increment. 
            # Since we need to handle up to large numbers, but input is small (54), simple loop works.
            
        if temp > 1:
            factors[temp] = 1
        
        coeff = 1
        radicand = 1
        
        for p in sorted(factors.keys()):
            e = factors[p]
            pairs = e // 2
            
            # Add to coefficient
            coeff *= (p ** pairs)
            
            # Remaining exponent goes to radicand
            remainder = e % 2
            if remainder == 1:
                radicand *= p
        
        return int(coeff), int(radicand)

    def compute_simplified(n):
        temp = n
        
        coeff = 1
        radicand = 1
        
        d = 2
        while d * d <= temp:
            if temp % d == 0:
                count = 0
                while temp % d == 0:
                    count += 1
                    temp //= d
                
                # Add pairs to coeff, remainder to radicand
                num_pairs = count // 2
                rem_exp = count % 2
                
                if num_pairs > 0:
                    coeff *= (d ** num_pairs)
                
                if rem_exp == 1:
                    radicand *= d
            
            d += 1
        
        # If temp > 1, it's a prime factor with exponent 1
        if temp > 1:
            radicand *= temp
            
        return coeff, int(radicand), int(temp)

        coeff = 1

            e = factors_count[p]

            # If odd exponent, multiply radicand by p

                radicand *= p
        
        return int(coeff), int(radicand)
