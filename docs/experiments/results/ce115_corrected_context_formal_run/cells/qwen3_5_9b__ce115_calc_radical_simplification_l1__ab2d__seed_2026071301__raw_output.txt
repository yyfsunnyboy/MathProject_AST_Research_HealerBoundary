def generate(level=1, **kwargs):
    import math
    
    radicand = kwargs.get("radicand", 27)
    
    # Factor out perfect squares from radicand to find square-free part and coefficient contribution
    temp_radicand = radicand
    coeff_contribution = 1
    
    d = 2
    while d * d <= temp_radicand:
        count = 0
        while temp_radicand % d == 0:
            count += 1
            temp_radicand //= d
        
        if count >= 2:
            # Extract pairs of factors as coefficient contribution
            num_pairs = count // 2
            coeff_contribution *= (d ** num_pairs)
    
    square_free_part = temp_radicand
    
    outer_coefficient = kwargs.get("outer_coefficient", 1)
    
    total_coefficient = outer_coefficient * coeff_contribution
    
    question_text = f"Rewrite {total_coefficient} * sqrt({radicand}) in simplest radical form as coefficient * sqrt(square_free_part)."

    return {
        "question_text": question_text,
        "correct_answer": {
            "coefficient": total_coefficient,
            "radicand": square_free_part
        },
        "oracle_payload": {"radicand": 27}
    }
