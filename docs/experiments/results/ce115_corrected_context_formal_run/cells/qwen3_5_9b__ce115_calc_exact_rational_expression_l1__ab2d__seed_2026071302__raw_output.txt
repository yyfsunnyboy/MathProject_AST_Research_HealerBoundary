def generate(level=1, **kwargs):
    from fractions import Fraction
    
    products = [{"left": "4.95", "right": "89.6", "sign": 1}, {"left": "-0.05", "right": "89.6", "sign": -1}]
    
    terms = []
    for term in products:
        left_val = Fraction(term["left"])
        right_val = Fraction(term["right"])
        sign = term["sign"]
        
        product_term = sign * left_val * right_val
        
        if len(terms) == 0:
            terms.append(product_term)
        else:
            last_sign, last_left, last_right = terms[-1]
            
            combined_product = (last_sign + sign) * left_val * right_val
            
            new_last_sign = last_sign + sign
            if new_last_sign != 0:
                terms.pop()
                terms.append((new_last_sign, left_val, right_val))
    
    total_sum = sum(terms[1][2] for _ in range(len(products)))

def generate(level=1, **kwargs):
    from fractions import Fraction
    
    products = [{"left": "4.95", "right": "89.6", "sign": 1}, {"left": "-0.05", "right": "89.6", "sign": -1}]
    
    total_sum = sum(term["sign"] * Fraction(term["left"]) * Fraction(term["right"]) for term in products)
    
    if total_sum.denominator == 1:
        answer_value = str(total_sum.numerator)
    else:
        common_divisor = gcd(abs(total_sum.numerator), abs(total_sum.denominator))
        
        numerator_reduced = total_sum.numerator // common_divisor
        denominator_reduced = total_sum.denominator // common_divisor
        
        if denominator_reduced < 0:
            numerator_reduced *= -1
            denominator_reduced *= -1
            
        answer_value = f"{numerator_reduced}/{denominator_reduced}"
