    candidates = [11, 12, 13, 14]
    n = 156
    
    def is_prime(num):
        """Check if a number is prime using native Python arithmetic."""
        if num < 2:
            return False
        for i in range(2, int(num ** 0.5) + 1):
            if num % i == 0:
                return False
        return True

    # Find the correct answer based on rules:
    # - Must be a frozen candidate
    # - Must divide n exactly (n % candidate == 0)
    # - Must be prime
    
    for cand in candidates:
        if n % cand == 0 and is_prime(cand):
            correct_answer = int(cand)
            break

    question_text = r"Find the unique integer $x$ from the set $\{11, 12, 13, 14\}$ such that $n \pmod x = 0$ and $x$ is a prime number. Given: $n = {n}$.".format(n=n)
    
    oracle_payload = {"candidates": candidates, "n": n}

    return {
        "question_text": question_text,
        "correct_answer": correct_answer,
        "oracle_payload": oracle_payload
    }
