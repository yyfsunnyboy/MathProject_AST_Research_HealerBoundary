        if is_prime(candidate) and (n % candidate == 0):
            correct_answer = int(candidate)

            latex_question = r"$\text{Select a prime factor of } n=156 \text{ from } \{{}{}\}$.".format(

            )

    return {
        "question_text": question_text,
        "correct_answer": correct_answer,
        "oracle_payload": oracle_payload
    }


def is_prime(num):
    if num < 2:
        return False
    for i in range(2, int(num**0.5) + 1):
        if num % i == 0:
            return False
    return True
