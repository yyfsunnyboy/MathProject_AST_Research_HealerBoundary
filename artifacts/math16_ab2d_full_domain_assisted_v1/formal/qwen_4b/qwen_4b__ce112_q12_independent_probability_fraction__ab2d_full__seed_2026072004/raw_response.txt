from core.prompts.domain_function_library import FractionOps


def generate(level=1, **kwargs):
    oracle_payload = kwargs.get("oracle_payload", {})
    
    p1_num = oracle_payload["p1"][0]
    p1_den = oracle_payload["p1"][1]
    a = FractionOps.from_parts(p1_num, p1_den)

    p2_num = oracle_payload["p2"][0]
    p2_den = oracle_payload["p2"][1]
    b = FractionOps.from_parts(p2_num, p2_den)

    value = FractionOps.mul(a, b)

    correct_answer = {
        "numerator": int(value.numerator),
        "denominator": int(value.denominator),
        "canonical_latex": str(FractionOps.to_exact(value)),
    }

    return {
        "question_text": oracle_payload["p1"][0] + "/" + str(oracle_payload["p1"][1]) + "\n" + oracle_payload["p2"][0] + "/" + str(oracle_payload["p2"][1]),
        "correct_answer": correct_answer,
        "oracle_payload": {
            "p1": [int(p1_num), int(p1_den)],
            "p2": [int(p2_num), int(p2_den)]
        }
    }
