```python
from core.prompts.domain_function_library import IntegerOps

def generate(level=1, **kwargs):
    oracle_payload = kwargs.get("oracle_payload", {"divisor_of": 216, "multiple_of": 18})
    
    divisor_of_val = oracle_payload["divisor_of"]
    multiple_of_val = oracle_payload["multiple_of"]

    divisors = IntegerOps.positive_divisors(divisor_of_val)
    
    count = sum(1 for d in divisors if IntegerOps.is_divisible(d, multiple_of_val))
    
    return {
        "question_text": f"有多少個正整數同時是 \(18\) 的倍數，也是 \(216\) 的因數？",
        "correct_answer": {"count": count},
        "oracle_payload": oracle_payload,
    }
```
