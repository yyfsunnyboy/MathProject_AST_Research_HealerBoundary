        m_d = int(x_coef_str.split('/')[1]) if '/' in x_coef_str else 1
        
        c_num = int(const_val_str.split('/')[0]) if '/' in const_val_str else int(const_val_str)
        c_den = int(const_val_str.split('/')[1]) if '/' in const_val_str else 1

        # Root is -c/m => -(num/den)/(m_n/m_d) = (-num * m_d) / (den * m_n)
        root_num = -(c_num * m_d)
        root_den = c_den * m_n
        
        final_roots_decimal.append(Decimal(root_num) / Decimal(root_den))

    for f in factors:
        x_coef_str = str(f["x_coefficient"])
        const_val_str = str(f["constant"]) if isinstance(f["constant"], (int, float)) else f["constant"]

        root_numerator = -(c_num * m_d)
        root_denominator = c_den * m_n
        
        sorted_roots.append((root_numerator, root_denominator))

    # Sort based on value: n/d < p/q <=> n*q < p*d (assuming positive denominators)

            c_num = int(const_val_str.split('/')[0])
            c_den = int(const_val_str.split('/')[1])
            
            val_num, val_den = c_num, c_d
